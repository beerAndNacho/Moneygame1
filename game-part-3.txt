  const ready = state.quest.progress >= quest.target && !state.quest.claimed;
  dom.questTitle.textContent = quest.title;
  dom.questDescription.textContent = state.quest.claimed
    ? '의뢰 완료 · 내일 새로운 의뢰가 도착합니다.'
    : `${targetSpecies.name} ${Math.min(state.quest.progress, quest.target)} / ${quest.target}마리`;
  dom.questReward.textContent = state.quest.claimed ? '완료' : formatWon(quest.reward);
  dom.questCard.classList.toggle('ready', ready);
  dom.questCard.classList.toggle('claimed', state.quest.claimed);
}

function renderMarket() {
  const value = state.inventory.reduce((sum, item) => sum + item.value, 0);
  dom.inventoryCount.textContent = `${state.inventory.length}마리`;
  dom.inventoryValue.textContent = formatWon(value);
  dom.sellAllButton.disabled = state.inventory.length === 0;
  dom.marketBadge.textContent = state.inventory.length;
  dom.marketBadge.classList.toggle('is-hidden', state.inventory.length === 0);
  dom.sortInventoryButton.textContent = inventorySort === 'value' ? '가치순' : '최근순';

  if (state.inventory.length === 0) {
    dom.inventoryList.innerHTML = `<div class="empty-state"><span class="empty-trap"></span><strong>보관함이 비어 있어요.</strong><p>항구에서 미끼와 낚시터를 고른 뒤 첫 통발을 내려보세요.</p></div>`;
    return;
  }

  const items = [...state.inventory].sort((a, b) => inventorySort === 'value'
    ? b.value - a.value
    : new Date(b.caughtAt) - new Date(a.caughtAt));

  dom.inventoryList.innerHTML = items.map((item) => {
    const species = speciesById(item.speciesId);
    return `<article class="inventory-card" style="${rarityStyle(item.rarity)};--species-color:${species.color}">
      <div class="species-art">${speciesIconSvg(species)}</div>
      <div class="inventory-copy">
        <small>${RARITY_META[item.rarity].label}</small>
        <strong>${item.name}</strong>
        <p>${item.length.toFixed(1)}cm · ${item.weight.toFixed(2)}kg</p>
      </div>
      <div class="inventory-actions">
        <b>${formatWon(item.value)}</b>
        <button type="button" data-sell="${item.uid}">판매</button>
      </div>
    </article>`;
  }).join('');
}

function gearBenefit(gearId, level) {
  if (gearId === 'trap') return `최대 어획 ${1 + Math.floor(level / 2)}~${2 + Math.floor(level / 2)}마리`;
  if (gearId === 'rope') return `실수 장력 피해 -${(level - 1) * 12}%`;
  return `희귀 유도 보너스 +${(level - 1) * 28}%`;
}

function renderGear() {
  dom.gearList.innerHTML = Object.values(GEAR).map((gear) => {
    const level = state.gear[gear.id];
    const maxed = level >= 5;
    const cost = getUpgradeCost(gear.id, level);
    const iconClass = gear.id === 'trap' ? 'trap-icon' : gear.id === 'rope' ? 'rope-icon' : 'lamp-icon';
    return `<article class="gear-card" style="--gear-accent:${gear.accent}">
      <div class="gear-card-top">
        <div class="gear-visual"><span class="mini-gear-icon ${iconClass}"></span></div>
        <div class="gear-copy"><small>${maxed ? '최고 등급' : '강화 가능'}</small><strong>${gear.name}</strong><p>${gear.description}</p></div>
        <span class="gear-level">Lv.${level}</span>
      </div>
      <div class="gear-progress"><i style="width:${level / 5 * 100}%"></i></div>
      <div class="gear-bottom">
        <span class="gear-benefit">${gearBenefit(gear.id, level)}</span>
        <button class="secondary-button" type="button" data-upgrade="${gear.id}" ${maxed || state.money < cost ? 'disabled' : ''}>${maxed ? '최대 강화' : `${formatWon(cost)} 강화`}</button>
      </div>
    </article>`;
  }).join('');
}

function renderCodex() {
  const discovered = SPECIES.filter((species) => state.codex[species.id]?.count > 0).length;
  const percent = Math.round(discovered / SPECIES.length * 100);
  dom.codexCount.textContent = discovered;
  dom.codexTotal.textContent = SPECIES.length;
  dom.codexPercent.textContent = `${percent}%`;
  dom.codexRing.style.setProperty('--progress', `${percent * 3.6}deg`);

  dom.codexGrid.innerHTML = SPECIES.map((species) => {
    const record = state.codex[species.id];
    const discoveredSpecies = Boolean(record?.count);
    return `<button class="codex-card ${discoveredSpecies ? '' : 'locked'}" type="button" data-species="${species.id}" style="${rarityStyle(species.rarity)};--species-color:${species.color}">
      ${record?.isNew ? '<span class="new-dot"></span>' : ''}
      <div class="species-art">${speciesIconSvg(species, !discoveredSpecies)}</div>
      <small>${discoveredSpecies ? RARITY_META[species.rarity].label : '미발견'}</small>
      <strong>${discoveredSpecies ? species.name : '???'}</strong>
    </button>`;
  }).join('');
}

function showScreen(name) {
  currentScreen = name;
  $$('.screen').forEach((screen) => screen.classList.toggle('active', screen.dataset.screen === name));
  $$('.nav-button').forEach((button) => button.classList.toggle('active', button.dataset.target === name));
  const fishing = name === 'fishing';
  dom.topBar.classList.toggle('is-hidden', fishing);
  dom.bottomNav.classList.toggle('is-hidden', fishing);
  if (!fishing) {
    const screen = $(`.screen[data-screen="${name}"] .scroll-content`);
    if (screen && screen.scrollTop < 2) screen.scrollTop = 0;
  }
  resizeCanvases();
}

function claimQuest() {
  ensureQuest();
  const quest = getDailyQuest(state.day);
  if (state.quest.claimed) {
    showToast('오늘 의뢰는 이미 완료했습니다.');
    return;
  }
  if (state.quest.progress < quest.target) {
    const species = speciesById(quest.speciesId);
    showToast(`${species.name}이 ${quest.target - state.quest.progress}마리 더 필요합니다.`);
    return;
  }
  state.quest.claimed = true;
  state.money += quest.reward;
  state.xp += 30;
  sound('coin');
  haptic([35, 35, 55]);
  saveState();
  renderAll();
  showToast(`의뢰 보상 ${formatWon(quest.reward)}을 받았습니다.`);
}

function restAtHarbor() {
  if (state.energy >= 100) {
    showToast('체력이 이미 가득합니다.');
    return;
  }
  state.energy = clamp(state.energy + 28, 0, 100);
  advanceTime(180);
  saveState();
  renderAll();
  showToast('항구에서 3시간 쉬어 체력을 회복했습니다.');
}

function startFishing() {
  const location = locationById(state.selectedLocation);
  const bait = baitById(state.selectedBait);
  if (currentLevel() < location.unlockLevel) {
    showToast(`선장 Lv.${location.unlockLevel}부터 출항할 수 있습니다.`);
    return;
  }
  if (state.energy < location.energy) {
    showToast('체력이 부족합니다. 번개 표시를 눌러 항구에서 쉬어보세요.');
    return;
  }
  if ((state.bait[bait.id] || 0) <= 0) {
    showToast('선택한 미끼가 없습니다.');
    return;
  }

  state.energy -= location.energy;
  state.bait[bait.id] -= 1;
  saveState();
  renderAll();

  session = {
    location,
    bait,
    phase: 'drop-ready',
    depth: 0,
    maxDepth: location.depth,
    dropping: false,
    stability: 100,
    tension: 0,
    cursor: 0.08,
    cursorDirection: 1,
    targetCenter: 0.62,
    targetWidth: 0.19,
    rounds: 0,
    perfect: 0,
    good: 0,
    miss: 0,
    combo: 0,
    bestCombo: 0,
    hazard: null,
    hazardUntil: 0,
    waitUntil: 0,
    inputLockUntil: 0,
    startedAt: performance.now(),
    refunded: false,
  };

  dom.fishingLocationLabel.textContent = location.name;
  setEvent('길게 눌러 원하는 수심까지 통발을 내려보세요.');
  updateFishingHud();
  showScreen('fishing');
  sound('splash');
}

function releaseTrap() {
  if (!session || !['dropping', 'drop-ready'].includes(session.phase)) return;
  session.dropping = false;
  dom.fishingActionButton.classList.remove('pressed');
  if (session.depth < Math.min(2.2, session.maxDepth * 0.32)) {
    session.phase = 'drop-ready';
    setEvent('조금 더 깊게 내려야 생물이 통발을 발견해요.', 'warning');
    updateFishingHud();
    return;
  }
  session.phase = 'soaking';
  session.waitUntil = performance.now() + 2300;
  setEvent('미끼 냄새가 바닷속으로 퍼지는 중…');
  updateFishingHud();
  sound('splash');
}

function beginRetrieve() {
  if (!session) return;
  session.phase = 'retrieve';
  session.targetCenter = 0.35 + Math.random() * 0.36;
  session.targetWidth = clamp(0.205 - (session.location.depth - 8) * 0.004 + (state.gear.rope - 1) * 0.006, 0.13, 0.22);
  setEvent('빛나는 구간에 맞춰 통발을 끌어올리세요.', 'success');
  updateFishingHud();
}

function pullTrap() {
  if (!session || session.phase !== 'retrieve') return;
  const now = performance.now();
  if (now < session.inputLockUntil) return;
  session.inputLockUntil = now + 120;

  if (session.hazard && now < session.hazardUntil) {
    session.miss += 1;
    session.combo = 0;
