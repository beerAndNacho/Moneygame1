function loadState() {
  const base = defaultState();
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (!raw) return base;
    const parsed = JSON.parse(raw);
    const merged = {
      ...base,
      ...parsed,
      bait: { ...base.bait, ...(parsed.bait || {}) },
      gear: { ...base.gear, ...(parsed.gear || {}) },
      codex: parsed.codex || {},
      records: parsed.records || {},
      inventory: Array.isArray(parsed.inventory) ? parsed.inventory : [],
      quest: { ...base.quest, ...(parsed.quest || {}) },
      settings: { ...base.settings, ...(parsed.settings || {}) },
    };
    if (!LOCATIONS.some((item) => item.id === merged.selectedLocation)) merged.selectedLocation = 'breakwater';
    if (!BAITS.some((item) => item.id === merged.selectedBait)) merged.selectedBait = 'squid';
    return merged;
  } catch (error) {
    console.warn('Could not load save data.', error);
    return base;
  }
}

function saveState() {
  try {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
  } catch (error) {
    console.warn('Could not save game state.', error);
  }
}

function currentLevel() {
  return levelFromXp(state.xp);
}

function weather() {
  return WEATHER[state.weatherIndex % WEATHER.length];
}

function periodLabel() {
  const hour = Math.floor(state.minutes / 60) % 24;
  if (hour >= 5 && hour < 10) return '아침';
  if (hour >= 10 && hour < 17) return '낮';
  if (hour >= 17 && hour < 21) return '저녁';
  return '밤';
}

function timeText() {
  const hour = Math.floor(state.minutes / 60) % 24;
  const minute = state.minutes % 60;
  return `${String(hour).padStart(2, '0')}:${String(minute).padStart(2, '0')}`;
}

function ensureQuest() {
  if (state.quest.day !== state.day) {
    state.quest = { day: state.day, progress: 0, claimed: false };
  }
}

function advanceTime(minutes) {
  state.minutes += Math.round(minutes);
  while (state.minutes >= 24 * 60) {
    state.minutes -= 24 * 60;
    state.day += 1;
    state.weatherIndex = (state.weatherIndex + 1) % WEATHER.length;
    state.energy = clamp(state.energy + 42, 0, 100);
  }
  ensureQuest();
}

function showToast(message, fishing = false) {
  clearTimeout(toastTimer);
  dom.toast.textContent = message;
  dom.toast.classList.toggle('fishing-toast', fishing);
  dom.toast.classList.add('show');
  toastTimer = window.setTimeout(() => dom.toast.classList.remove('show'), 2200);
}

function ensureAudio() {
  if (!audioContext) {
    const AudioContextClass = window.AudioContext || window.webkitAudioContext;
    if (AudioContextClass) audioContext = new AudioContextClass();
  }
  if (audioContext?.state === 'suspended') audioContext.resume().catch(() => {});
}

function sound(kind = 'tap') {
  if (!state.settings.sound) return;
  ensureAudio();
  if (!audioContext) return;

  const oscillator = audioContext.createOscillator();
  const gain = audioContext.createGain();
  const now = audioContext.currentTime;
  const config = {
    tap: [310, 0.035, 'square', 0.025],
    perfect: [740, 0.13, 'sine', 0.07],
    good: [470, 0.09, 'triangle', 0.05],
    miss: [120, 0.12, 'sawtooth', 0.04],
    coin: [880, 0.14, 'square', 0.045],
    upgrade: [540, 0.22, 'triangle', 0.06],
    splash: [180, 0.18, 'sine', 0.05],
  }[kind] || [310, 0.05, 'sine', 0.03];

  oscillator.type = config[2];
  oscillator.frequency.setValueAtTime(config[0], now);
  if (kind === 'coin' || kind === 'upgrade' || kind === 'perfect') {
    oscillator.frequency.exponentialRampToValueAtTime(config[0] * 1.55, now + config[1]);
  } else if (kind === 'miss' || kind === 'splash') {
    oscillator.frequency.exponentialRampToValueAtTime(Math.max(70, config[0] * 0.55), now + config[1]);
  }
  gain.gain.setValueAtTime(config[3], now);
  gain.gain.exponentialRampToValueAtTime(0.0001, now + config[1]);
  oscillator.connect(gain).connect(audioContext.destination);
  oscillator.start(now);
  oscillator.stop(now + config[1] + 0.02);
}
