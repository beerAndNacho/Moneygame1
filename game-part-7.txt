    ctx.moveTo(w * 0.08, h * 0.48);
    ctx.quadraticCurveTo(w * 0.31, h * 0.39, w * 0.58, h * 0.48);
    ctx.quadraticCurveTo(w * 0.34, h * 0.55, w * 0.08, h * 0.48);
    ctx.fill();
    ctx.fillStyle = '#5f1f29';
    ctx.fillRect(w * 0.22, h * 0.46, 4, 3);
    ctx.restore();
  }

  const trapX = w * 0.52;
  const normalizedDepth = session.maxDepth ? session.depth / session.maxDepth : 0;
  const trapY = surface + 34 + normalizedDepth * (bottomY - surface - 64);
  ctx.strokeStyle = session.tension > 72 ? '#e8b76b' : '#9a724b';
  ctx.lineWidth = 2.5;
  ctx.beginPath();
  ctx.moveTo(trapX, surface - 4);
  const ropeBend = Math.sin(t * 1.4) * (4 + weather().wind * 10) + session.tension * 0.035;
  ctx.quadraticCurveTo(trapX + ropeBend, (surface + trapY) / 2, trapX, trapY - 17);
  ctx.stroke();
  drawTrap(ctx, trapX, trapY, session.tension, t);

  if (session.phase === 'dropping') {
    ctx.fillStyle = 'rgba(174, 238, 226, 0.16)';
    ctx.beginPath();
    ctx.arc(trapX, trapY + 10, 22 + Math.sin(t * 8) * 3, 0, Math.PI * 2);
    ctx.fill();
  }
}

function drawFishSilhouette(ctx, x, y, size, direction) {
  ctx.save();
  ctx.translate(x, y);
  ctx.scale(direction * size, size);
  ctx.globalAlpha = 0.17;
  ctx.fillStyle = '#011922';
  ctx.beginPath();
  ctx.ellipse(0, 0, 10, 5, 0, 0, Math.PI * 2);
  ctx.fill();
  ctx.beginPath();
  ctx.moveTo(-9, 0);
  ctx.lineTo(-17, -7);
  ctx.lineTo(-17, 7);
  ctx.closePath();
  ctx.fill();
  ctx.restore();
}

function drawTrap(ctx, x, y, tension, t) {
  const sway = Math.sin(t * 2.3) * (1 + tension * 0.02);
  ctx.save();
  ctx.translate(x + sway, y);
  ctx.fillStyle = '#182b33';
  ctx.strokeStyle = '#a1c9c2';
  ctx.lineWidth = 2;
  ctx.fillRect(-18, -13, 36, 27);
  ctx.strokeRect(-18, -13, 36, 27);
  ctx.strokeStyle = 'rgba(151, 201, 194, 0.6)';
  ctx.lineWidth = 1;
  for (let line = -12; line <= 12; line += 6) {
    ctx.beginPath();
    ctx.moveTo(line, -12);
    ctx.lineTo(line, 13);
    ctx.stroke();
  }
  for (let line = -7; line <= 7; line += 7) {
    ctx.beginPath();
    ctx.moveTo(-17, line);
    ctx.lineTo(17, line);
    ctx.stroke();
  }
  ctx.fillStyle = '#e1c278';
  ctx.fillRect(-4, -3, 8, 6);
  ctx.restore();
}

function updateFrame(timestamp) {
  const dt = Math.min(0.05, (timestamp - lastFrame) / 1000);
  lastFrame = timestamp;

  if (currentScreen === 'home') drawHarbor(timestamp);
  if (currentScreen === 'fishing' && session) {
    updateFishing(dt, timestamp);
    drawOcean(timestamp);
  }
  requestAnimationFrame(updateFrame);
}

function updateFishing(dt, now) {
  if (!session) return;
  if (session.phase === 'dropping' && session.dropping) {
    const dropSpeed = session.maxDepth / 3.25;
    session.depth = clamp(session.depth + dt * dropSpeed, 0, session.maxDepth);
    if (session.depth >= session.maxDepth) releaseTrap();
  }

  if (session.phase === 'soaking' && now >= session.waitUntil) beginRetrieve();

  if (session.phase === 'retrieve') {
    const speed = 0.56 + session.location.depth * 0.018 + session.rounds * 0.018;
    session.cursor += session.cursorDirection * speed * dt;
    if (session.cursor >= 1) {
      session.cursor = 1;
      session.cursorDirection = -1;
    } else if (session.cursor <= 0) {
      session.cursor = 0;
      session.cursorDirection = 1;
    }
    session.tension = clamp(session.tension - dt * (2.8 + state.gear.rope * 0.5), 0, 100);
    if (session.hazard && now >= session.hazardUntil) {
      session.hazard = null;
      setEvent('물살이 잦아들었습니다. 다시 타이밍을 노리세요.');
    }
  }
  updateFishingHud();
}

function initialize() {
  ensureQuest();
  setupEvents();
  renderAll();
  resizeCanvases();
  showScreen('home');
  requestAnimationFrame(updateFrame);

  if ('serviceWorker' in navigator) {
    window.addEventListener('load', () => {
      navigator.serviceWorker.register('/service-worker.js').catch((error) => console.warn('Service worker registration failed.', error));
    });
  }
}

initialize();
