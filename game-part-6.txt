  const w = dom.harborCanvas.clientWidth || 430;
  const h = dom.harborCanvas.clientHeight || 850;
  if (!homeStars.length) {
    homeStars = Array.from({ length: 62 }, (_, index) => ({
      x: pseudo(index * 17 + 3) * w,
      y: pseudo(index * 31 + 11) * h * 0.46,
      size: 0.7 + pseudo(index * 7 + 5) * 1.7,
      phase: pseudo(index * 13 + 9) * Math.PI * 2,
    }));
  }
  if (!oceanParticles.length) {
    oceanParticles = Array.from({ length: 46 }, (_, index) => ({
      x: pseudo(index * 23 + 5),
      y: pseudo(index * 41 + 7),
      speed: 0.004 + pseudo(index * 11 + 2) * 0.008,
      size: 0.7 + pseudo(index * 19 + 1) * 1.8,
    }));
    fishSilhouettes = Array.from({ length: 11 }, (_, index) => ({
      x: pseudo(index * 29 + 2),
      y: 0.32 + pseudo(index * 37 + 3) * 0.48,
      size: 0.55 + pseudo(index * 17 + 8) * 0.75,
      speed: 0.009 + pseudo(index * 13 + 6) * 0.013,
      direction: index % 2 ? 1 : -1,
    }));
  }
}

function pseudo(seed) {
  const value = Math.sin(seed * 12.9898) * 43758.5453;
  return value - Math.floor(value);
}

function drawHarbor(timestamp) {
  if (currentScreen !== 'home') return;
  const { context: ctx, width: w, height: h } = resizeCanvas(dom.harborCanvas);
  const t = timestamp * 0.001;
  const horizon = h * 0.42;
  ctx.clearRect(0, 0, w, h);

  const sky = ctx.createLinearGradient(0, 0, 0, horizon + 80);
  sky.addColorStop(0, '#050c18');
  sky.addColorStop(0.55, '#10243c');
  sky.addColorStop(1, '#6b5a62');
  ctx.fillStyle = sky;
  ctx.fillRect(0, 0, w, horizon + 80);

  const starStrength = weather().star;
  for (const star of homeStars) {
    const alpha = (0.28 + Math.sin(t * 1.4 + star.phase) * 0.18) * starStrength;
    ctx.fillStyle = `rgba(222,242,234,${clamp(alpha, 0.1, 0.8)})`;
    ctx.fillRect(Math.round(star.x), Math.round(star.y), star.size, star.size);
  }

  ctx.fillStyle = 'rgba(242, 218, 149, 0.9)';
  ctx.beginPath();
  ctx.arc(w * 0.79, h * 0.16, 24, 0, Math.PI * 2);
  ctx.fill();
  ctx.fillStyle = '#0b1728';
  ctx.beginPath();
  ctx.arc(w * 0.805, h * 0.151, 24, 0, Math.PI * 2);
  ctx.fill();

  ctx.fillStyle = '#1a2b38';
  ctx.beginPath();
  ctx.moveTo(0, horizon + 10);
  ctx.lineTo(w * 0.16, horizon - 34);
  ctx.lineTo(w * 0.31, horizon + 2);
  ctx.lineTo(w * 0.48, horizon - 47);
  ctx.lineTo(w * 0.65, horizon - 5);
  ctx.lineTo(w * 0.82, horizon - 29);
  ctx.lineTo(w, horizon + 4);
  ctx.lineTo(w, horizon + 80);
  ctx.lineTo(0, horizon + 80);
  ctx.closePath();
  ctx.fill();

  const sea = ctx.createLinearGradient(0, horizon, 0, h);
  sea.addColorStop(0, '#1b5d69');
  sea.addColorStop(0.35, '#0d4051');
  sea.addColorStop(1, '#071923');
  ctx.fillStyle = sea;
  ctx.fillRect(0, horizon, w, h - horizon);

  for (let row = 0; row < 6; row += 1) {
    ctx.strokeStyle = `rgba(136, 221, 216, ${0.16 - row * 0.018})`;
    ctx.lineWidth = 1;
    ctx.beginPath();
    const y = horizon + row * 17 + Math.sin(t * 0.8 + row) * 2;
    for (let x = -20; x <= w + 20; x += 10) {
      const waveY = y + Math.sin(x * 0.045 + t * (0.8 + weather().wind) + row) * 2.2;
      if (x === -20) ctx.moveTo(x, waveY); else ctx.lineTo(x, waveY);
    }
    ctx.stroke();
  }

  ctx.fillStyle = '#78828a';
  for (let x = -10; x < w * 0.41; x += 12) {
    const rockY = horizon - 1 + Math.sin(x * 0.2) * 3;
    ctx.fillRect(x, rockY, 14, 9);
    ctx.fillStyle = x % 24 ? '#5d6a73' : '#879098';
  }
  ctx.fillStyle = '#707b84';
  for (let x = w * 0.7; x < w + 10; x += 13) {
    ctx.fillRect(x, horizon + Math.sin(x * 0.21) * 3, 15, 10);
    ctx.fillStyle = x % 26 ? '#5a6871' : '#849097';
  }

  const boatBob = Math.sin(t * 1.8) * 3;
  drawPixelBoat(ctx, w * 0.53, horizon - 5 + boatBob, 0.92);

  const lx = w * 0.86;
  ctx.fillStyle = '#e8e5d5';
  ctx.fillRect(lx - 12, horizon - 89, 24, 78);
  ctx.fillStyle = '#c94f43';
  ctx.fillRect(lx - 15, horizon - 92, 30, 17);
  ctx.fillRect(lx - 15, horizon - 52, 30, 14);
  ctx.fillStyle = '#26333a';
  ctx.fillRect(lx - 8, horizon - 107, 16, 15);
  ctx.fillStyle = '#efdc82';
  ctx.fillRect(lx - 6, horizon - 103, 12, 8);
  ctx.fillStyle = `rgba(242, 223, 132, ${0.18 + Math.sin(t * 2.4) * 0.06})`;
  ctx.beginPath();
  ctx.moveTo(lx, horizon - 101);
  ctx.lineTo(w * 0.42, horizon - 67);
  ctx.lineTo(w * 0.42, horizon - 127);
  ctx.closePath();
  ctx.fill();
}

function drawOcean(timestamp) {

  const { context: ctx, width: w, height: h } = resizeCanvas(dom.oceanCanvas);
  const t = timestamp * 0.001;
  const surface = h * 0.205;
  ctx.clearRect(0, 0, w, h);

  const sky = ctx.createLinearGradient(0, 0, 0, surface + 15);
  sky.addColorStop(0, '#061020');
  sky.addColorStop(1, '#39566b');
  ctx.fillStyle = sky;
  ctx.fillRect(0, 0, w, surface + 25);

  for (let i = 0; i < 22; i += 1) {
    const x = pseudo(i * 19 + 4) * w;
    const y = pseudo(i * 13 + 8) * surface * 0.78;
    ctx.fillStyle = `rgba(231,243,232,${0.18 + pseudo(i * 7 + 2) * 0.35})`;
    ctx.fillRect(x, y, 1.2, 1.2);
  }

  ctx.fillStyle = '#263844';
  ctx.beginPath();
  ctx.moveTo(0, surface + 5);
  ctx.lineTo(w * 0.15, surface - 22);
  ctx.lineTo(w * 0.31, surface + 3);
  ctx.lineTo(w * 0.47, surface - 34);
  ctx.lineTo(w * 0.65, surface - 4);
  ctx.lineTo(w * 0.82, surface - 27);
  ctx.lineTo(w, surface + 1);
  ctx.lineTo(w, surface + 30);
  ctx.lineTo(0, surface + 30);
  ctx.fill();

  const water = ctx.createLinearGradient(0, surface, 0, h);
  water.addColorStop(0, '#2c8190');
  water.addColorStop(0.34, '#135368');
  water.addColorStop(0.7, '#0a3043');
  water.addColorStop(1, '#061923');
  ctx.fillStyle = water;
  ctx.fillRect(0, surface, w, h - surface);

  ctx.save();
  ctx.globalAlpha = 0.09;
  for (let ray = 0; ray < 5; ray += 1) {
    ctx.fillStyle = '#b9f0e4';
    ctx.beginPath();

    ctx.moveTo(x, surface);
    ctx.lineTo(x + 115, h);
    ctx.lineTo(x + 168, h);
    ctx.lineTo(x + 31, surface);
    ctx.fill();
  }
  ctx.restore();

  ctx.strokeStyle = 'rgba(189, 238, 228, 0.48)';
  ctx.lineWidth = 1.4;
  ctx.beginPath();
  for (let x = -8; x <= w + 8; x += 8) {
    const y = surface + Math.sin(x * 0.06 + t * 2.1) * 3;
    if (x === -8) ctx.moveTo(x, y); else ctx.lineTo(x, y);
  }
  ctx.stroke();

  drawPixelBoat(ctx, w * 0.52, surface - 6 + Math.sin(t * 1.8) * 2.7, 0.72);

  const bottomY = h * 0.79;
  ctx.fillStyle = '#263a3f';
  ctx.beginPath();
  ctx.moveTo(0, bottomY + 7);
  for (let x = 0; x <= w; x += 16) {
    ctx.lineTo(x, bottomY + Math.sin(x * 0.08) * 7 + pseudo(x + 5) * 10);
  }
  ctx.lineTo(w, h);
  ctx.lineTo(0, h);
  ctx.closePath();
  ctx.fill();

  for (let i = 0; i < 50; i += 1) {
    const x = pseudo(i * 18 + 4) * w;
    const y = bottomY + pseudo(i * 27 + 9) * (h - bottomY);
    const size = 3 + pseudo(i * 7 + 11) * 10;
    ctx.fillStyle = i % 3 === 0 ? '#61716e' : i % 3 === 1 ? '#435759' : '#33494d';
    ctx.fillRect(Math.round(x), Math.round(y), Math.round(size), Math.round(size * 0.65));
  }

    particle.y -= particle.speed * (1 + weather().wind);
    if (particle.y < 0.23) particle.y = 0.92;
    const x = (particle.x * w + Math.sin(t * 0.4 + particle.y * 8) * 9 + w) % w;
    const y = particle.y * h;
    ctx.fillStyle = 'rgba(195, 231, 224, 0.22)';
    ctx.fillRect(x, y, particle.size, particle.size);
  }

  for (const fish of fishSilhouettes) {
    fish.x += fish.speed * fish.direction * 0.01;
    if (fish.x > 1.1) fish.x = -0.1;
    if (fish.x < -0.1) fish.x = 1.1;
    drawFishSilhouette(ctx, fish.x * w, fish.y * h, fish.size, fish.direction);
  }
