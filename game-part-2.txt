function shadeColor(hex, amount) {
  const clean = hex.replace('#', '');
  const number = Number.parseInt(clean, 16);
  const r = clamp((number >> 16) + amount, 0, 255);
  const g = clamp(((number >> 8) & 0xff) + amount, 0, 255);
  const b = clamp((number & 0xff) + amount, 0, 255);
  return `#${[r, g, b].map((part) => Math.round(part).toString(16).padStart(2, '0')).join('')}`;
}

function renderTopBar() {
  const level = currentLevel();
  dom.dayLabel.textContent = `${state.day}일차 · ${periodLabel()}`;
  dom.levelLabel.textContent = `선장 Lv.${level}`;
  dom.moneyLabel.textContent = Math.round(state.money).toLocaleString('ko-KR');
  dom.energyLabel.textContent = Math.round(state.energy);
  dom.weatherLabel.textContent = `${weather().name} · ${timeText()}`;
  dom.soundButton.textContent = state.settings.sound ? '♪' : '×';
  dom.soundButton.setAttribute('aria-label', state.settings.sound ? '소리 끄기' : '소리 켜기');
  dom.soundToggle.checked = state.settings.sound;
  dom.hapticsToggle.checked = state.settings.haptics;
}

function renderLocationOptions() {
  const level = currentLevel();
  dom.locationOptions.innerHTML = LOCATIONS.map((location) => {
    const locked = level < location.unlockLevel;
    const selected = state.selectedLocation === location.id;
    return `<button class="location-card ${selected ? 'selected' : ''} ${locked ? 'locked' : ''}" type="button" data-location="${location.id}" style="--spot-accent:${location.accent}" ${locked ? 'aria-disabled="true"' : ''}>
      <span>${locked ? `Lv.${location.unlockLevel}` : `${location.depth}m`}</span>
      <small>${location.subtitle}</small>
      <strong>${location.name}</strong>
    </button>`;
  }).join('');
}

function renderBaitOptions() {
  dom.baitOptions.innerHTML = BAITS.map((bait) => {
    const count = state.bait[bait.id] || 0;
    return `<button class="bait-card ${state.selectedBait === bait.id ? 'selected' : ''}" type="button" data-bait="${bait.id}" ${count <= 0 ? 'disabled' : ''}>
      <span class="bait-pixel-icon ${bait.icon}" aria-hidden="true"></span>
      <p><strong>${bait.name}</strong><small>${bait.note}</small><b>보유 ${count}</b></p>
    </button>`;
  }).join('');
}

function renderDeparture() {
  const location = locationById(state.selectedLocation);
  const baitCount = state.bait[state.selectedBait] || 0;
  dom.tripEnergy.textContent = `⚡ ${location.energy}`;
  dom.trapSummary.textContent = `접이식 Lv.${state.gear.trap}`;
  dom.ropeSummary.textContent = `코팅 Lv.${state.gear.rope}`;
  dom.lampSummary.textContent = `심해 Lv.${state.gear.lamp}`;
  dom.departSubtitle.textContent = `${location.name} · 수심 ${location.depth}m`;
  dom.departButton.disabled = state.energy < location.energy || baitCount <= 0;
}
