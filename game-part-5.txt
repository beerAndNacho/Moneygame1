      dom.actionLabel.textContent = session.hazard && performance.now() < session.hazardUntil ? '지금은 기다려!' : '당기기';
      dom.actionHint.textContent = '빛나는 타이밍 구간을 노리세요';
      break;
    case 'complete':
      dom.roundLabel.textContent = '완료';
      dom.actionLabel.textContent = '조업 결과 확인 중';
      dom.actionHint.textContent = '통발을 열어보고 있습니다';
      break;
    default:
      break;
  }
}

function sellOne(uid) {
  const index = state.inventory.findIndex((item) => item.uid === uid);
  if (index < 0) return;
  const [item] = state.inventory.splice(index, 1);
  state.money += item.value;
  saveState();
  renderAll();
  sound('coin');
  haptic(25);
  showToast(`${item.name} 판매 · ${formatWon(item.value)}`);
}

function sellAll() {
  if (!state.inventory.length) return;
  const total = state.inventory.reduce((sum, item) => sum + item.value, 0);
  const count = state.inventory.length;
  state.inventory = [];
  state.money += total;
  saveState();
  renderAll();
  sound('coin');
  haptic([30, 30, 60]);
  showToast(`${count}마리를 판매해 ${formatWon(total)}을 받았습니다.`);
}

function upgradeGear(gearId) {
  const level = state.gear[gearId];
  if (level >= 5) return;
  const cost = getUpgradeCost(gearId, level);
  if (state.money < cost) {
    showToast(`${formatWon(cost - state.money)}이 더 필요합니다.`);
    return;
  }
  state.money -= cost;
  state.gear[gearId] += 1;
  state.xp += 18;
  saveState();
  renderAll();
  sound('upgrade');
  haptic([30, 30, 45]);
  showToast(`${GEAR[gearId].name}을 Lv.${state.gear[gearId]}로 강화했습니다.`);
}

function openSpeciesDetail(speciesId) {
  const species = speciesById(speciesId);
  const codex = state.codex[speciesId];
  if (!codex?.count) {
    showToast('아직 발견하지 못한 생물입니다.');
    return;
  }
  const record = state.records[speciesId] || {};
  state.codex[speciesId].isNew = false;
  saveState();
  dom.detailIcon.style.setProperty('--species-color', species.color);
  dom.detailIcon.innerHTML = speciesIconSvg(species);
  dom.detailRarity.textContent = RARITY_META[species.rarity].label;
  dom.detailRarity.style.color = RARITY_COLORS[species.rarity];
  dom.detailName.textContent = species.name;
  dom.detailDescription.textContent = species.description;
  dom.detailLength.textContent = record.maxLength ? `${record.maxLength.toFixed(1)}cm` : '-';
  dom.detailWeight.textContent = record.maxWeight ? `${record.maxWeight.toFixed(2)}kg` : '-';
  dom.detailValue.textContent = record.maxValue ? formatWon(record.maxValue) : '-';
  dom.detailCount.textContent = `${record.count || codex.count}회`;
  renderCodex();
  openModal(dom.detailModal);
}

      pullTrap();
    }
  });

  const releaseHandler = (event) => {
    if (session?.dropping) releaseTrap();
    if (event?.pointerId !== undefined) {
      try { dom.fishingActionButton.releasePointerCapture?.(event.pointerId); } catch (_) { /* no-op */ }
    }
  };
  dom.fishingActionButton.addEventListener('pointerup', releaseHandler);
  dom.fishingActionButton.addEventListener('pointercancel', releaseHandler);
  dom.fishingActionButton.addEventListener('lostpointercapture', () => {

  });

  });
  dom.hapticsToggle.addEventListener('change', () => {

  });

    const confirmed = window.confirm('모든 조업 기록과 장비 진행을 초기화할까요?');
    if (!confirmed) return;

    dom.installButton.classList.add('is-hidden');

    event.preventDefault();

  });

  window.addEventListener('resize', resizeCanvases, { passive: true });
  document.addEventListener('visibilitychange', () => {
    if (document.hidden && session?.dropping) releaseTrap();
    saveState();
  });
}

function resizeCanvas(canvas) {
  const rect = canvas.getBoundingClientRect();
  const dpr = Math.min(window.devicePixelRatio || 1, 2);
  const width = Math.max(1, Math.round(rect.width * dpr));
  const height = Math.max(1, Math.round(rect.height * dpr));
  if (canvas.width !== width || canvas.height !== height) {
    canvas.width = width;
    canvas.height = height;
  }
  const context = canvas.getContext('2d');
  context.setTransform(dpr, 0, 0, dpr, 0, 0);
  context.imageSmoothingEnabled = false;
  return { context, width: rect.width, height: rect.height, dpr };
}
